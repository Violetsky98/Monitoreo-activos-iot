CREATE TABLE IF NOT EXISTS activos (
    rfid_uid TEXT PRIMARY KEY,
    nombre_activo TEXT,
    tipo TEXT
);

CREATE TABLE IF NOT EXISTS historial_activos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    rfid_uid TEXT,
    usuario TEXT,
    fecha_hora TEXT,
    estado TEXT,
    ubicacion TEXT,
    FOREIGN KEY (rfid_uid) REFERENCES activos (rfid_uid)
);
